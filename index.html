.github/workflows/study_reminder.yml 
name: Recordatorio Diario FocusDev
on:
  schedule:
    - cron: '0 0 * * *' # 8:00 PM Venezuela (UTC-4)
  workflow_dispatch:

jobs:
  notify:
    runs-on: ubuntu-latest
    steps:
      - name: Consultar Supabase y enviar Telegram
        env:
          SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
          SUPABASE_KEY: ${{ secrets.SUPABASE_KEY }}
          TG_BOT_TOKEN: ${{ secrets.TG_BOT_TOKEN }}
          TG_CHAT_ID: ${{ secrets.TG_CHAT_ID }}
        run: |
          python3 -c "
          import os, urllib.request, json
          url = f'{os.environ[\"SUPABASE_URL\"]}/rest/v1/tasks?completed=eq.false&select=title,due_date&order=due_date.asc&limit=4'
          req = urllib.request.Request(url, headers={'apikey': os.environ['SUPABASE_KEY'], 'Authorization': f'Bearer {os.environ[\"SUPABASE_KEY\"]}'})
          tasks = json.loads(urllib.request.urlopen(req).read().decode())
          resumen = '\n'.join([f'• {t[\"title\"]} ({t[\"due_date\"]})' for t in tasks]) or 'Sin tareas pendientes.'
          msg = f'⚡ FocusDev: Hora de estudiar hoy.\n\nPróximos plazos:\n{resumen}'
          tg_url = f'https://api.telegram.org/bot{os.environ[\"TG_BOT_TOKEN\"]}/sendMessage'
          data = json.dumps({'chat_id': os.environ['TG_CHAT_ID'], 'text': msg}).encode()
          urllib.request.urlopen(urllib.request.Request(tg_url, data=data, headers={'Content-Type': 'application/json'}))
          "
