1. Script PostgreSQL (Pegar en Supabase SQL Editor)

Crea las 4 tablas y configura el bucket público deliverables para que puedas subir archivos directamente desde tu teléfono o laptop. 

-- 1. Tabla de Secciones / Módulos
CREATE TABLE IF NOT EXISTS modules (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title TEXT NOT NULL,
    category TEXT NOT NULL,
    target_date DATE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Tabla de Tareas Autoasignadas
CREATE TABLE IF NOT EXISTS tasks (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    module_id BIGINT NOT NULL REFERENCES modules(id) ON DELETE CASCADE,
    title TEXT NOT NULL,
    due_date DATE NOT NULL,
    priority TEXT DEFAULT 'Media',
    completed BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Tabla de Entregables con URL del archivo en Storage
CREATE TABLE IF NOT EXISTS deliverables (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    task_id BIGINT REFERENCES tasks(id) ON DELETE SET NULL,
    title TEXT NOT NULL,
    due_date DATE NOT NULL,
    submitted_at TIMESTAMPTZ DEFAULT NOW(),
    file_name TEXT NOT NULL,
    file_url TEXT,
    repo_url TEXT,
    notes TEXT,
    on_time BOOLEAN NOT NULL
);

-- 4. Tabla de Check-in Diario de Estudio
CREATE TABLE IF NOT EXISTS study_logs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    log_date DATE NOT NULL UNIQUE,
    hours NUMERIC(4,2) NOT NULL,
    summary TEXT NOT NULL
);

-- Desactivar RLS para uso personal directo
ALTER TABLE modules DISABLE ROW LEVEL SECURITY;
ALTER TABLE tasks DISABLE ROW LEVEL SECURITY;
ALTER TABLE deliverables DISABLE ROW LEVEL SECURITY;
ALTER TABLE study_logs DISABLE ROW LEVEL SECURITY;

-- Crear bucket público 'deliverables' y política de subida/lectura
INSERT INTO storage.buckets (id, name, public)
VALUES ('deliverables', 'deliverables', true)
ON CONFLICT (id) DO NOTHING;

CREATE POLICY "Acceso Publico Personal Deliverables"
ON storage.objects FOR ALL
USING (bucket_id = 'deliverables')
WITH CHECK (bucket_id = 'deliverables');

2. Bot de Recordatorio Diario en GitHub Actions (Opcional)

Guárdalo como .github/workflows/study_reminder.yml en tu repositorio para recibir alertas en tu Telegram si no has estudiado o si tienes un entregable próximo. 

name: Recordatorio Diario FocusDev
on:
  schedule:
    - cron: '0 0 * * *' # 8:00 PM hora Venezuela (UTC-4)
  workflow_dispatch:

jobs:
  notify:
    runs-on: ubuntu-latest
    steps:
      - name: Consultar Supabase y enviar alerta a Telegram
        env:
          SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
          SUPABASE_KEY: ${{ secrets.SUPABASE_KEY }}
          TG_BOT_TOKEN: ${{ secrets.TG_BOT_TOKEN }}
          TG_CHAT_ID: ${{ secrets.TG_CHAT_ID }}
        run: |
          python3 -c "
          import os, urllib.request, json
          url = f'{os.environ[\"SUPABASE_URL\"]}/rest/v1/tasks?completed=eq.false&select=title,due_date&order=due_date.asc&limit=3'
          req = urllib.request.Request(url, headers={'apikey': os.environ['SUPABASE_KEY'], 'Authorization': f'Bearer {os.environ[\"SUPABASE_KEY\"]}'})
          tasks = json.loads(urllib.request.urlopen(req).read().decode())
          resumen = '\n'.join([f'• {t[\"title\"]} (Vence: {t[\"due_date\"]})' for t in tasks]) or 'Sin tareas pendientes.'
          msg = f'⚡ FocusDev: Hora de tu sesión de estudio de hoy.\n\nPróximos entregables:\n{resumen}'
          tg_url = f'https://api.telegram.org/bot{os.environ[\"TG_BOT_TOKEN\"]}/sendMessage'
          data = json.dumps({'chat_id': os.environ['TG_CHAT_ID'], 'text': msg}).encode()
          urllib.request.urlopen(urllib.request.Request(tg_url, data=data, headers={'Content-Type': 'application/json'}))
          "
